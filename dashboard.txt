@st.cache_data
def cargar(ruta):
    ext = ruta.rsplit(".", 1)[-1].lower()
    if ext in ("xlsx", "xls"):
        df = pd.read_excel(ruta, engine="openpyxl")
    else:
        for enc in ["utf-8-sig", "utf-8", "latin-1", "cp1252"]:
            try:
                df = pd.read_csv(ruta, encoding=enc)
                break
            except (UnicodeDecodeError, UnicodeError):
                continue
        else:
            df = pd.read_csv(ruta, encoding="latin-1", errors="replace")
    return df


def cargar_bytes(uploaded_file):
    nombre = uploaded_file.name.lower()
    if nombre.endswith(".xlsx") or nombre.endswith(".xls"):
        df = pd.read_excel(uploaded_file, engine="openpyxl")
    else:
        for enc in ["utf-8-sig", "utf-8", "latin-1", "cp1252"]:
            try:
                uploaded_file.seek(0)
                df = pd.read_csv(uploaded_file, encoding=enc)
                break
            except (UnicodeDecodeError, UnicodeError):
                continue
        else:
            uploaded_file.seek(0)
            df = pd.read_csv(uploaded_file, encoding="latin-1", errors="replace")
    return df


def normalizar_columnas(df, mapeo):
    """Renombra columnas segun el mapeo confirmado por el usuario."""
    renombrar = {}
    for campo_std, col_original in mapeo.items():
        if col_original and col_original in df.columns and col_original != campo_std:
            renombrar[col_original] = campo_std
    df = df.rename(columns=renombrar)
    # Convertir numericos
    for col in ["lat", "lng", "rating", "num_resenas", "score_potencial_argos"]:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors="coerce")
    if "score_potencial_argos" in df.columns:
        df["score_potencial_argos"] = df["score_potencial_argos"].fillna(0)
    return df

def detectar_columnas(headers):
    """Detecta automaticamente que columna corresponde a cada campo."""
    headers_lower = {h: h.lower().replace("_", " ").replace("-", " ") for h in headers}
    mapeo = {}

    for campo, config in CAMPOS_ESTANDAR.items():
        mapeo[campo] = None
        for candidato in config["candidatos"]:
            # Match exacto primero
            for h, h_low in headers_lower.items():
                if h_low == candidato or h == campo:
                    mapeo[campo] = h
                    break
            if mapeo[campo]:
                break
            # Match parcial
            for h, h_low in headers_lower.items():
                if candidato in h_low:
                    mapeo[campo] = h
                    break
            if mapeo[campo]:
                break

    return mapeo

        while True:
            linea = proceso.stdout.readline()
            if not linea and proceso.poll() is not None:

with tab6:

    GROQ_KEY_CHAT = os.getenv("GROQ_API_KEY", "")
    if not GROQ_KEY_CHAT:
        st.warning("GROQ_API_KEY no encontrada en .env")
    else:
        def contexto_chat(df):
            t = len(df)
            if t == 0:
                return "Dataset vacio."
            top_c = df["ciudad"].value_counts().head(5).to_string()
            alta_n = (df.get("vende_cemento_probable", pd.Series()) == "Alta").sum()
            media_n = (df.get("vende_cemento_probable", pd.Series()) == "Media").sum()
            comp_n = (df.get("tipo_cliente_argos", pd.Series()) == "Cliente de competencia").sum()
            prio_n = (df.get("score_potencial_argos", pd.Series(dtype=float)) >= 70).sum()
            construrama_n = df.get("nombre_original", pd.Series()).str.lower().str.contains("construrama", na=False).sum()
            score_avg = round(df.get("score_potencial_argos", pd.Series(dtype=float)).mean(), 1) if "score_potencial_argos" in df.columns else 0
            sin_t = (df["telefono"].fillna("") == "").sum()

            if "score_potencial_argos" in df.columns:
                top5 = df.nlargest(5, "score_potencial_argos")[
                    [c for c in ["nombre_comercial_limpio", "ciudad", "score_potencial_argos", "vende_cemento_probable"]
                     if c in df.columns]
                ].to_string(index=False)
            else:
                top5 = "No disponible"

            return f"""
Total: {t} ferreterias | Municipios: {df['ciudad'].nunique()}
Alta prob. cemento: {alta_n} | Media: {media_n}
Competencia: {comp_n} | Construrama: {construrama_n}
Prospectos score>=70: {prio_n} | Score prom: {score_avg}
Sin telefono: {sin_t}
Top municipios: {top_c}
Top 5 prospectos: {top5}"""

        def chat_groq(historial, contexto):
            r = requests.post(
                "https://api.groq.com/openai/v1/chat/completions",
                headers={
                    "Authorization": f"Bearer {GROQ_KEY_CHAT}",
                    "Content-Type": "application/json"
                },
                json={
                    "model": "llama-3.3-70b-versatile",
                    "messages": [
                        {
                            "role": "system",
                            "content": f"Eres asistente de inteligencia comercial para Cementos Argos Colombia. "
                                       f"Responde en espanol, conciso, con datos del censo.\n\n"
                                       f"DATOS:\n{contexto}"
                        }
                    ] + historial,
                    "temperature": 0.3,
                    "max_tokens": 600
                },
                timeout=30
            )
            r.raise_for_status()
            return r.json()["choices"][0]["message"]["content"]

        if "historial_chat" not in st.session_state:
            st.session_state.historial_chat = []

        st.markdown("**Preguntas rapidas:**")
        ce1, ce2, ce3 = st.columns(3)
        with ce1:
            if st.button("Alta probabilidad cemento?"):
                st.session_state.pregunta_rapida = "Cuantas ferreterias tienen alta probabilidad de vender cemento?"

                st.session_state.pregunta_rapida = "Cuales son los 5 mejores prospectos para visitar?"

        for msg in st.session_state.historial_chat:
            with st.chat_message(msg["role"]):
                st.write(msg["content"])

        pregunta_default = st.session_state.pop("pregunta_rapida", "") if "pregunta_rapida" in st.session_state else ""
        pregunta = st.chat_input("Escribe tu pregunta sobre el censo...")
        pregunta_final = pregunta or pregunta_default

            with st.chat_message("user"):
                st.write(pregunta_final)
            st.session_state.historial_chat.append({"role": "user", "content": pregunta_final})

            with st.chat_message("assistant"):
                with st.spinner("Analizando..."):
                    try:
                        ctx = contexto_chat(df)
                        resp = chat_groq(st.session_state.historial_chat, ctx)
                        st.write(resp)
                        st.session_state.historial_chat.append({"role": "assistant", "content": resp})
                    except Exception as e:
                        st.error(f"Error: {str(e)[:100]}")

        if st.session_state.historial_chat:
            if st.button("Limpiar conversacion"):
                st.session_state.historial_chat = []

    if df_argos_raw is not None:

        def _find_col_argos(cols, keywords):
            cols_lower = {c: c.lower().replace("_"," ").replace("-"," ") for c in cols}
            for c, cl in cols_lower.items():
                if any(k in cl for k in keywords):
                    return c
            return None

        col_nombre_argos = _find_col_argos(df_argos_raw.columns, [
            "nombre de la cuenta","nombre cuenta","razon social","cuenta",
            "empresa","negocio","establecimiento","nombre"
        ])

        col_ciudad_a  = _find_col_argos(df_argos_raw.columns, ["poblaci","ciudad","municipio"])

                    "colombia","antioquia","zomac","sa","grupo"

                def norm_c(texto):
                    if not isinstance(texto, str): return ""
                    t = texto.lower()
                    for a,b in {chr(225):"a",chr(233):"e",chr(237):"i",
                                 chr(243):"o",chr(250):"u",chr(241):"n"}.items():
                        t = t.replace(a, b)
                    t = _re.sub(r"[^a-z\s]", " ", t)
                    t = _re.sub(r"\s+", " ", t).strip()
                    return " ".join(p for p in t.split()
                                    if p not in STOPWORDS_C and len(p) > 1)

                    ta, tb = set(a.split()), set(b.split())
                    if not ta or not tb: return 0.0
                    return len(ta & tb) / len(ta | tb)

                    d2 = _re.sub(r"\D","", str(t2 or ""))[-8:]
                    return bool(d1) and bool(d2) and d1 == d2

                df_res["lat"] = pd.to_numeric(df_res["lat"], errors="coerce")

                conocidos = df_res[df_res["estado_cruce"] == "Cliente Argos conocido"]
